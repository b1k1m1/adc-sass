PRIORITY 1: ADD DATABASE INDEXES
================================================================================

Create migration:
  php artisan make:migration add_critical_indexes_for_multiuser

Then add this code:

public function up(): void
{
    // members - most queried table
    Schema::table('members', function (Blueprint $table) {
        $table->index('Trans_flag', 'idx_members_trans_flag');
        $table->index(['Trans_flag', 'ParticipantId'], 'idx_members_type_id');
    });

    // partattendence - attendance reports (your heaviest queries)
    Schema::table('partattendence', function (Blueprint $table) {
        $table->index('CalenderDate', 'idx_attend_date');
        $table->index(['ParticipantId', 'CalenderDate'], 'idx_attend_part_date');
        $table->index(['CalenderDate', 'AttendanceYN'], 'idx_attend_date_yn');
    });

    // claims
    Schema::table('claims', function (Blueprint $table) {
        $table->index(['status', 'claim_no'], 'idx_claims_status');
        $table->index(['ParticipantId', 'service_from'], 'idx_claims_part_date');
    });
}

Run:
  php artisan migrate

Verify indexes:
  SHOW INDEX FROM members;
  SHOW INDEX FROM partattendence;
  SHOW INDEX FROM claims;

PRIORITY 2: CLEAN UP TRIM() QUERIES
================================================================================

Run once to clean existing data:
  UPDATE members SET ParticipantId = TRIM(ParticipantId);
  UPDATE partattendence SET ParticipantId = TRIM(ParticipantId);
  UPDATE claims SET ParticipantId = TRIM(ParticipantId);

Then change all queries from:
  ->whereRaw('TRIM(ParticipantId) = ?', [$participantId])

To:
  ->where('ParticipantId', trim($participantId))

PRIORITY 3: ELOQUENT MODELS
================================================================================

Create app/Models/Member.php:
---------------
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\HasMany;

class Member extends Model
{
    protected $table = 'members';
    protected $primaryKey = 'ParticipantId';
    public $incrementing = false;
    protected $keyType = 'string';
    public $timestamps = false;
    protected $guarded = [];
    
    public function attendances(): HasMany
    {
        return $this->hasMany(Attendance::class, 'ParticipantId', 'ParticipantId');
    }
    
    public function claims(): HasMany
    {
        return $this->hasMany(Claim::class, 'ParticipantId', 'ParticipantId');
    }
}
